        questions.forEach((qObj, qIndex) => {

            const selected = document.querySelector(`input[name="q${qIndex}"]:checked`);

           

            for(let i=0; i<4; i++){

                document.getElementById(`opt-container-${qIndex}-${i}`).classList.remove('correct', 'incorrect');

            }

           

            document.getElementById(`opt-container-${qIndex}-${qObj.ans}`).classList.add('correct');

           

            if (selected) {

                const userAns = parseInt(selected.value);

                if (userAns === qObj.ans) {

                    score++;

                } else {

                    document.getElementById(`opt-container-${qIndex}-${userAns}`).classList.add('incorrect');

                }

            }

           

            document.getElementById(`loigiai-${qIndex}`).style.display = 'block';

        });

 

        const resultBox = document.getElementById('result-box');

        resultBox.style.display = 'block';

        const total = questions.length;

        const percentage = Math.round((score/total)*100)/10; // Điểm hệ 10, làm tròn 1 chữ số

       

        resultBox.innerHTML = `Hoàn thành bài thi! <br> Số câu đúng: ${score} / ${total} <br> Điểm của bạn: <span style="color:var(--primary-color); font-size:1.5em;">${percentage}</span>`;

       

        window.scrollTo({ top: 0, behavior: 'smooth' });

       

        document.getElementById('submit-btn').disabled = true;

        document.getElementById('submit-btn').innerText = "Đã nộp bài";

        document.getElementById('submit-btn').style.backgroundColor = "#9aa0a6";

    }

 

    window.onload = renderQuiz;

</script>
